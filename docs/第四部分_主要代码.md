## 5）新建、打开、保存和另存为

```cpp
void MainWindow::newFile()
{
    if (!maybeSave())
        return;
    ui->editor->clear();
    assignDocumentPath(QString());
    refreshWindowTitle();
}

void MainWindow::openFile()
{
    const QString path = QFileDialog::getOpenFileName(
        this, tr("打开文件"), QString(),
        tr("文本文件 (*.txt);;所有文件 (*.*)"));
    if (!path.isEmpty())
        readDocument(path);
}
```

文件读写集中由 `TextFileService` 处理：

```cpp
const TextFileLoadResult loaded =
```

```cpp
template <typename Handler>

                      QPlainTextEdit *editor,
```

```cpp
{
    const QTextCursor cursor = ui->editor->textCursor();

        tr("length: %1  lines: %2  Ln: %3  Col: %4")
            .arg(ui->editor->toPlainText().size())
```

## 9）Cut、Copy、Paste 的使能状态

```cpp
void MainWindow::refreshActionAvailability()
{
    const bool selected =
        ui->editor->textCursor().hasSelection();
    const QMimeData *clipboardData =
        QApplication::clipboard()->mimeData();

    ui->actionCut->setEnabled(selected);
    ui->actionCopy->setEnabled(selected);
    ui->actionPaste->setEnabled(
        clipboardData && clipboardData->hasText());
}
```

## 11）替换窗口操作

```cpp
void MainWindow::substituteEverywhere(const QString &text,
                                      const QString &replacement,
                                      bool caseSensitive)
{
    const int count = SearchService::substituteAll(
        ui->editor->document(), text,
        replacement, caseSensitive);

    if (count == 0) {
        QMessageBox::information(
            this, tr("替换全部"),
            tr("未找到“%1”。").arg(text));
        return;
    }

    statusBar()->showMessage(
        tr("已替换 %1 处").arg(count), 3000);
}
```
